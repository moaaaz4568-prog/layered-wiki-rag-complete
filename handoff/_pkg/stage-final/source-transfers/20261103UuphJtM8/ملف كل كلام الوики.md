def safe_filename(name, max_len=120):
    name = re.sub(r'[\\/*?\":<>|\n\r\t]', "", name).strip()
    return (name or "video")[:max_len]

def clean_vtt(text):
    text = re.sub(r"WEBVTT.*?\n\n",                      "", text, flags=re.DOTALL)
    text = re.sub(r"NOTE.*?\n\n",                        "", text, flags=re.DOTALL)
    text = re.sub(r"\d{2}:\d{2}:\d{2}\.\d+\s*-->.*\n", "", text)
    text = re.sub(r"\d{2}:\d{2}\.\d+\s*-->.*\n",       "", text)
    text = re.sub(r"<[^>]+>",                            "", text)
    text = re.sub(r"^\s*\d+\s*$",                       "", text, flags=re.MULTILINE)
    lines = [l.strip() for l in text.splitlines() if l.strip()]
    deduped = []
    for l in lines:
        if not deduped or l != deduped[-1]:
            deduped.append(l)
    return " ".join(deduped)

def try_yt_dlp(video_id, cookie_path, temp_dir, use_pot=False):

        "ignoreerrors": True,
        "format": "best",
        "check_formats": False,
        "user_agent": UA,
        "http_headers": {
            "User-Agent": UA,

        },
    }
    if use_pot:
        dl_opts["extractor_args"] = {"youtube": {"player_client": ["web", "web_safari"]}}
    try:
        with yt_dlp.YoutubeDL(dl_opts) as ydl:
            ydl.download([f"https://www.youtube.com/watch?v={video_id}"])

    vtt_files = glob.glob(os.path.join(temp_dir, f"{video_id}*.vtt"))

        vtt_files = glob.glob(os.path.join(temp_dir, f"{video_id}.*"))
        vtt_files = [v for v in vtt_files if v.endswith((".vtt", ".ttml", ".srv1", ".srv2", ".srv3", ".json3"))]

            return None, None, "no subtitle files produced"
    chosen = None

    for lang in LANG_PREF:

        if chosen: break
    if not chosen:

                chosen, chosen_lang = vf, "?"
                break
    if not chosen:
        return None, None, "all subtitle files were live_chat"
    try:
        with open(chosen, "r", encoding="utf-8") as f:
            raw = f.read()
        cleaned = clean_vtt(raw)

def try_yta(video_id, cookies_dict):

            NoTranscriptFound, VideoUnavailable, CouldNotRetrieveTranscript)
    except ImportError:
        return None, None, "youtube_transcript_api not installed"

        "User-Agent": UA,

        "Referer": "https://www.youtube.com/",
    })
    try:
        api = YouTubeTranscriptApi(http_client=session)
        transcript_list = api.list(video_id)
        try:
            t = transcript_list.find_manually_created_transcript(LANG_PREF)

        return text, t.language_code, None

        return None, None, f"yta error: {type(e).__name__}: {str(e)[:80]}"

def try_innertube(video_id, cookies_dict):
    if not cookies_dict.get("SAPISID"): return None, None, "no SAPISID cookie"
    auth = sapisid_hash(cookies_dict["SAPISID"])
    session = requests.Session()
    session.cookies.update(cookies_dict)
    session.headers.update({
        "User-Agent": UA, "Accept-Language": "en-US,en;q=0.9",
        "Origin": "https://www.youtube.com", "Referer": "https://www.youtube.com/",
        "X-Youtube-Client-Name": "1", "X-Youtube-Client-Version": INNERTUBE_CLIENT_VER,
        "Content-Type": "application/json",
    })
    payload = {
        "context": {"client": {"clientName": "WEB", "clientVersion": INNERTUBE_CLIENT_VER, "hl":"en","gl":"US"}},
        "videoId": video_id,
    }
    try:
        r = session.post(f"https://www.youtube.com/youtubei/v1/next?key={INNERTUBE_API_KEY}",
                         json=payload, headers={"Authorization": auth}, timeout=30)
        if r.status_code != 200: return None, None, f"next HTTP {r.status_code}"
        data = r.json()
    except Exception as e: return None, None, f"next error: {type(e).__name__}: {str(e)[:80]}"
    params_list = []
    def walk(o):
        if isinstance(o, dict):
            for k, v in o.items():
                if k == "getTranscriptEndpoint" and isinstance(v, dict):
                    if v.get("params"): params_list.append(v["params"])
                walk(v)
        elif isinstance(o, list):
            for it in o: walk(it)
    walk(data)
    if not params_list: return None, None, "no getTranscriptEndpoint"
    for params in params_list:
        try:
            r = session.post(
                f"https://www.youtube.com/youtubei/v1/get_transcript?key={INNERTUBE_API_KEY}",
                json={
                    "context": {"client": {"clientName": "WEB", "clientVersion": INNERTUBE_CLIENT_VER, "hl":"en","gl":"US"}},
                    "params": params,
                },
                headers={"Authorization": auth, "X-Goog-AuthUser": "0"},
                timeout=30,
            )
            if r.status_code != 200: continue
            tdata = r.json()
            segments = []
            def find_seg(o):
                if isinstance(o, dict):
                    if "transcriptSegmentRenderer" in o:
                        seg = o["transcriptSegmentRenderer"]
                        runs = seg.get("snippet", {}).get("runs", [])
                        segments.append("".join(r.get("text", "") for r in runs))
                    for v in o.values(): find_seg(v)
                elif isinstance(o, list):
                    for it in o: find_seg(it)
            find_seg(tdata)
            if segments: return " ".join(segments), "?", None
        except Exception: continue
    return None, None, "get_transcript returned no segments"

def extract_urls_from_endpoint(target_url, cookie_path):
    opts = {
        "cookiefile": cookie_path, "quiet": True, "no_warnings": True,
        "ignoreerrors": True, "extract_flat": True,
        "http_headers": {"User-Agent": UA, "Accept-Language": "en-US,en;q=0.9"},
    }
    try:
        with yt_dlp.YoutubeDL(opts) as ydl:
            info = ydl.extract_info(target_url, download=False)
        if not info: return []
        entries = [e for e in (info.get("entries") or [info]) if e]
        vids = []
        for e in entries:
            vid = e.get("id")
            title = e.get("title", "")
            if vid and len(str(vid)) >= 11:
                vids.append((vid, title))
        return vids
    except Exception as e:
        print(f"⚠️ Error loading {target_url}: {e}")
        return []

def process_one_video(video_id, video_title, channel_tag, cookie_path, cookies_dict,
                      temp_dir, output_dir, pot_available=False):
    result = {"video_id": video_id, "title": video_title, "channel": channel_tag,
              "status": "failed", "strategy": None, "lang": None, "length": 0, "error": None}

    text, lang, err = try_yt_dlp(video_id, cookie_path, temp_dir, use_pot=False)
    if text and text.strip():
        result.update({"status": "ok", "strategy": "yt_dlp", "lang": lang, "length": len(text), "error": None})
        _save_transcript(text, video_id, video_title, channel_tag, output_dir)
        return result
    strat1_err = err

    if pot_available:
        text, lang, err = try_yt_dlp(video_id, cookie_path, temp_dir, use_pot=True)
        if text and text.strip():
            result.update({"status": "ok", "strategy": "yt_dlp+pot", "lang": lang, "length": len(text), "error": None})
            _save_transcript(text, video_id, video_title, channel_tag, output_dir)
            return result
        strat1b_err = err
    else:
        strat1b_err = "pot server not running"

    text, lang, err = try_yta(video_id, cookies_dict)
    if text and text.strip():
        result.update({"status": "ok", "strategy": "yta", "lang": lang, "length": len(text), "error": None})
        _save_transcript(text, video_id, video_title, channel_tag, output_dir)
        return result
    strat2_err = err

    text, lang, err = try_innertube(video_id, cookies_dict)
    if text and text.strip():
        result.update({"status": "ok", "strategy": "innertube", "lang": lang, "length": len(text), "error": None})
        _save_transcript(text, video_id, video_title, channel_tag, output_dir)
        return result
    strat3_err = err

    result["error"] = f"s1:{strat1_err} | s1b:{strat1b_err} | s2:{strat2_err} | s3:{strat3_err}"
    return result

def _save_transcript(text, video_id, title, channel_tag, output_dir):
    base = f"{channel_tag}__{safe_filename(title)}__{video_id}.txt"
    path = os.path.join(output_dir, base)
    if os.path.exists(path):
        i = 1
        while os.path.exists(path):
            path = os.path.join(output_dir, f"{channel_tag}__{safe_filename(title)}__{video_id}_{i}.txt")
            i += 1
    with open(path, "w", encoding="utf-8") as f:
        f.write(text)

**أولاً: RAG - Retrieval-Augmented Generation**

RAG مش موديل، دي معمارية استرجاع قبل التوليد. خطواتها كالتالي:

- **Ingestion والتقطيع Chunking:** النص الخام يتقسم لوحدات دلالية، مش تقطيع عشوائي بالحروف. الوحدة المثالية هي الفقرة أو القسم تحت عنوان، مع overlap بسيط عشان السياق ما يتقطعش عند الحدود. التقطيع السيئ هو أول سبب لفشل الاسترجاع، لأن المقطع اللي اتقطع من عنوانه أو جدوله بيبقى غير مفهوم لوحده.
- **Embedding:** كل مقطع بيتحول لمتجه عددي Vector في فضاء دلالي، عن طريق موديل تضمين. النصوص المتشابهة في المعنى بتبقى قريبة في الفضاء ده، حتى لو الكلمات مختلفة. القياس الشائع هو Cosine Similarity.
- **الفهرسة Indexing:** المتجهات بتتخزن في قاعدة فيكتور Vector Database، زي ChromaDB، مع Metadata لكل مقطع: المصدر، العنوان، الشخصية، التاريخ.
- **الاسترجاع Retrieval:** سؤال المستخدم بيتحول لنفس الفضاء بنفس موديل التضمين، وبيتم استرجاع أقرب k مقطع.
- **إعادة الترتيب Reranking:** المقاطع المسترجعة بتمر على Cross-Encoder يقارن السؤال مع كل مقطع مباشرة، ويرتبهم بدقة أعلى من التشابه الأولي.
- **التوليد Generation:** الموديل اللغوي بياخد المقاطع دي كسياق، ويجاوب منها فقط، مع استشهاد بمصدر كل مقطع.

نقاط قوة RAG علمياً: التأسيس Grounding، يعني الإجابة مربوطة بمصدر قابل للتحقق، والحداثة Freshness، لأن أي مستند جديد يدخل الفهرس ويبقى قابل للاسترجاع من غير إعادة تدريب، وخطأ الاسترجاع في استعلام واحد لا يلوث باقي النظام.

نقاط ضعفه: تجزئة السياق، والضوضاء لما top-k يكبر، وضعف البحث الدلالي وحده في الأسماء والمصطلحات الدقيقة، وده سبب إضافة البحث بالكلمات.

**ثانياً: البحث الهجين Hybrid Retrieval**

الحل العلمي للضعف ده هو دمج نوعين استرجاع:
- **Sparse Retrieval، زي BM25:** بحث بالكلمة بالظبط، قوي في الأسماء والعناوين والمصطلحات.
- **Dense Retrieval:** بحث بالمعنى عن طريق المتجهات، قوي في الصياغات المختلفة لنفس الفكرة.
- **Fusion، زي Reciprocal Rank Fusion:** دمج ترتيب القائمتين في ترتيب واحد، وبعدين Reranking.

وفوق ده كله: **Metadata Filtering.** الفلتر هنا مش تحسين، ده قيد منطقي. في حالتنا الشخصية هي Metadata إجباري، مش وزن في التشابه، لأن التشابه وحده لا يميز بين مصدرين متشابهين الموضوع مختلفين المؤلف.

**ثالثاً: LLM Wiki - الذاكرة المترجمة**

فكرة Karpathy هي تحويل المعرفة الخام إلى ذاكرة دلالية مترجمة Compiled Semantic Memory. الموديل اللغوي بيقرأ المصادر، ويكتب صفحات منظمة هرمياً، بعناوين وروابط، وملف توجيه Router يحدد مكان كل نوع معلومة، وبعدين الاستعلام يبدأ من الفهرس وينزل للصفحات المطلوبة بس، وده اسمه Progressive Disclosure.

نقاط قوتها: ضغط دلالي، يعني المعرفة المتكررة بتتلخص مرة واحدة، وتنقل أسهل، وتكلفة استدلال أقل في الاستعلامات المتكررة، لأن الإجابة الجاهزة أقرب من إعادة قراءة الخام كل مرة.

نقاط ضعفها، ودي مثبتة في النقد العلمي: فقدان معلومة وقت الاستخلاص Information Loss، لأن التلخيص بيسقط تفاصيل نادرة، وتضاعف الخطأ Error Compounding، لأن صفحة خاطئة بتبقى مرجع لصفحات تانية، وقدم المحتوى Staleness، لأن الصفحة المترجمة بتفضل مقروءة بثقة حتى بعد ما مصدرها يتغير، وتكلفة التحديث، لأن تغيير واحد ممكن يفرض إعادة معالجة لصفحات مرتبطة، وغياب Provenance لو الصفحة مش مربوطة بمصدرها الخام.

الخلاصة العلمية: الويكي طبقة تجميع، مش طبقة حقيقة. الحقيقة تفضل في الخام.

**رابعاً: GraphRAG - استرجاع بالجراف**

GraphRAG بيضيف طبقة علاقات فوق RAG. الموديل بيستخرج كيانات Entities وعلاقات Relations من كل مقطع، ويبني جراف، وبعدين بيعمل اكتشاف مجتمعات Community Detection وتلخيص هرمي لكل مجتمع.

نقاط قوته: الأسئلة متعددة القفزات Multi-hop، زي العلاقات والتأثيرات والموضوعات العابرة للمستندات، اللي البحث المسطح بالتشابه بيفشل فيها.

تكلفته: استدعاء موديل لكل مقطع للاستخراج، وقاعدة جراف إضافية للصيانة، وتعقيد في التحديث. لذلك علمياً هو طبقة لاحقة، مش طبقة استرجاع أولى.

**خامساً: OKF - Open Knowledge Format**

OKF مش أداة، ده عقد تنسيق Standard Format. حزمة Bundle هي مجلد ملفات Markdown عادية، كل مفهوم في ملف مستقل، هوية المفهوم هي مساره، وفي Front Matter حقل `type` إجباري، والروابط بين الملفات هي الجراف، وملف `index.md` للاكتشاف التدريجي. الهدف هو قابلية النقل Portability: نفس البيانات تتقري من أي موديل أو أداة، من غير قاعدة بيانات ملكية أو SDK في طريق القراءة.

**سادساً: هنعمل إيه بالظبط، بالمصطلحات دي**

- **هنعمل RAG كأساس:** لكل شخصية Collection مستقلة في ChromaDB، بتضمين محلي متعدد اللغات bge-m3، بأبعاد 1024، وتقطيع دلالي على العناوين والفقرات، وMetadata لكل مقطع: الشخصية، مسار الملف، العنوان.
- **هنعمل استرجاع هجين:** BM25 زائد Dense زائد Reranking، مع فلتر شخصية إجباري، وTop-k محدود، وتوسيع للصفحة الكاملة في الأسئلة الشمولية.
- **هنحتفظ بالويكي كطبقة مترجمة:** للتنقل والشرح المستقر، لكن الإجابة النهائية من مقاطع خام مسترجعة، مش من صفحة الويكي وحدها، وكل إجابة باستشهاد لمسار خام.
- **هنعمل Router وIndex لكل شخصية:** ملف توجيه يحدد مكان كل نوع معلومة، والاستعلام يبدأ من الفهرس قبل فتح الصفحات.
- **هنعمل تقييم Evaluation قبل التوسع:** اختبار Known-Answer، يعني سؤال إجابته معروفة في صفحة معينة، ونقيس Recall@k، هل الصفحة الصح ظهرت في النتائج، وهل الإجابة فيها استشهاد صحيح.
- **هنعمل فهرسة تزايدية Incremental Indexing:** المادة الجديدة تدخل الخام الأول، وبعدين تتقطع وتتفهرس لوحدها، من غير إعادة بناء كاملة.
- **هنأجل GraphRAG:** لمرحلة الأسئلة العلاقية فقط، بعد ما الاسترجاع الأساسي يثبت نجاحه.
- **هنأجل أنظمة ذاكرة الفرق:** لأنها تحل مشكلة الكتابة المتزامنة والصلاحيات، ودي مش موجودة عند مستخدم واحد.

**ليه الهجين هو الحل العلمي؟**
لأن RAG لوحده يفقد التنظيم والتلخيص، والويكي لوحده يفقد التأسيس والحداثة، والجراف لوحده يفقد البساطة والتكلفة. الهجين بياخد من كل طبقة وظيفتها: الخام للتأسيس، والويكي للتنقل، والبحث الهجين للاسترجاع، والجراف لاحقاً للعلاقات.
